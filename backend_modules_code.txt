from odoo import models, fields, api

class Erpv6TenantMixin(models.AbstractModel):
    _name = 'erpv6.tenant.mixin'
    _description = 'Multi-tenant Mixin'

    root_tenant_id = fields.Many2one('res.company', string='Root Tenant', required=True, default=lambda self: self.env.company)
    
    @api.model
    def create(self, vals_list):
        if isinstance(vals_list, dict):
            vals_list = [vals_list]
        for vals in vals_list:
            if 'root_tenant_id' not in vals:
                vals['root_tenant_id'] = self.env.company.id
        return super().create(vals_list)

    def _search(self, args, offset=0, limit=None, order=None, count=False):
        # Auto-filter by tenant unless sudo or specific context
        if not self.env.su and 'erpv6_bypass_tenant' not in self.env.context:
            args += [('root_tenant_id', '=', self.env.company.id)]
        return super()._search(args, offset=offset, limit=limit, order=order, count=count)

from odoo import models, fields, api, _
from odoo.exceptions import UserError
import json
import base64
from cryptography.hazmat.primitives.ciphers.aead import AESGCM
import os

class Erpv6KbModule(models.Model):
    _name = 'erpv6.kb.module'
    _inherit = ['erpv6.tenant.mixin']
    _description = 'Encrypted Knowledge Module'

    module_id = fields.Char(string='Module ID', required=True, index=True)
    category = fields.Char(string='Category')
    content_encrypted = fields.Binary(string='Encrypted Content')
    is_locked = fields.Boolean(string='Locked (V6 Method)', default=False)
    version = fields.Char(string='Version')
    
    # Key management (in production use env vars or secure vault)
    _encryption_key = os.environ.get('ERPv6_KEY', '0123456789abcdef0123456789abcdef0123456789abcdef0123456789abcdef')

    def write(self, vals):
        if self.is_locked and not self.env.user.has_group('erpv6_core.group_v6_admin'):
            raise UserError(_("Cannot modify locked V6 Method modules."))
        return super().write(vals)

    def set_content(self, content_dict):
        """Encrypts and saves content"""
        aesgcm = AESGCM(base64.b16decode(self._encryption_key.upper()))
        data = json.dumps(content_dict).encode()
        nonce = os.urandom(12)
        ct = aesgcm.encrypt(nonce, data, None)
        self.content_encrypted = base64.b64encode(nonce + ct)

    def get_content(self):
        """Decrypts and returns content"""
        if not self.content_encrypted:
            return {}
        aesgcm = AESGCM(base64.b16decode(self._encryption_key.upper()))
        raw = base64.b64decode(self.content_encrypted)
        nonce, ct = raw[:12], raw[12:]
        pt = aesgcm.decrypt(nonce, ct, None)
        return json.loads(pt.decode())

# models/typst_document.py
from odoo import models, fields, api
import requests
import json

class Erpv6TypstDocument(models.Model):
    _name = 'erpv6.typst.document'
    
    project_id = fields.Many2one('erpv6.project', required=True)
    template_id = fields.Many2one('erpv6.typst.template', required=True)
    state = fields.Selection([('draft','Draft'),('rendering','Rendering'),('ready','Ready'),('sent','Sent')], default='draft')
    pdf_file = fields.Binary(string='PDF')
    disc_target = fields.Char(related='project_id.disc_profile')
    
    def action_generate(self):
        self.state = 'rendering'
        # Prepare data from project + KB prompts
        data_payload = {
            'template': self.template_id.template_code,
            'data': self.project_id.collected_data_json,
            'disc': self.disc_target,
            'prompts': self._get_v6_prompts()
        }
        
        # Call external Typst Microservice
        try:
            resp = requests.post('http://typst-service:8000/render', json=data_payload, timeout=30)
            if resp.status_code == 200:
                self.pdf_file = base64.b64encode(resp.content)
                self.state = 'ready'
        except Exception as e:
            self.state = 'draft'
            raise UserError(f"Generation failed: {str(e)}")

    def _get_v6_prompts(self):
        # Fetch locked prompts from KB
        return []
